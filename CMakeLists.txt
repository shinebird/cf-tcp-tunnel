cmake_minimum_required(VERSION 3.21)
project(cf_tcp_tunnel_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost 1.83 REQUIRED COMPONENTS json)
find_package(OpenSSL REQUIRED)
add_executable(cf-tunnel src/main.cpp)
target_link_libraries(cf-tunnel PRIVATE Boost::json OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
    target_compile_definitions(cf-tunnel PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(cf-tunnel PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
    target_compile_options(cf-tunnel PRIVATE /W4 /EHsc /bigobj)
else()
    target_compile_options(cf-tunnel PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS cf-tunnel DESTINATION bin)
install(FILES $<TARGET_PDB_FILE:cf-tunnel> DESTINATION bin OPTIONAL)
