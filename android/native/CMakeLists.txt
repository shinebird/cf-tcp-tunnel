cmake_minimum_required(VERSION 3.22)
project(cf_tunnel_native LANGUAGES C CXX)
# Also optimize native code when Studio selects the debug APK variant.
if(ANDROID)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Native build profile" FORCE)
    message(STATUS "CF native profile: ${CMAKE_BUILD_TYPE}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)

if(ANDROID)
    if(NOT CF_NATIVE_DEPS)
        message(FATAL_ERROR "CF_NATIVE_DEPS is required; first run tools/prepare-native.ps1 or tools/prepare-native.sh")
    endif()
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(CF_TRIPLET arm64-android-cftunnel)
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(CF_TRIPLET x64-android-cftunnel)
    else()
        message(FATAL_ERROR "This port builds arm64-v8a and x86_64 only")
    endif()
    # Dependencies are built directly from local source with the same NDK/API/ABI. Do not
    # replace AGP's toolchain with a desktop toolchain during Gradle configure.
    set(prefix "${CF_NATIVE_DEPS}/${CF_TRIPLET}")
    if(NOT EXISTS "${prefix}/include/openssl/ssl.h" OR NOT EXISTS "${prefix}/lib/libssl.a" OR NOT EXISTS "${prefix}/lib/libcrypto.a")
        message(FATAL_ERROR "Missing Android native dependencies in ${prefix}; Windows .lib files cannot be used")
    endif()
    list(PREPEND CMAKE_FIND_ROOT_PATH "${prefix}")
    list(PREPEND CMAKE_PREFIX_PATH "${prefix}")
    set(OPENSSL_ROOT_DIR "${prefix}")
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    # Pin imported archives to this ABI prefix; never locate a desktop installation.
    add_library(OpenSSL::Crypto STATIC IMPORTED)
    set_target_properties(OpenSSL::Crypto PROPERTIES IMPORTED_LOCATION "${prefix}/lib/libcrypto.a" INTERFACE_INCLUDE_DIRECTORIES "${prefix}/include")
    add_library(OpenSSL::SSL STATIC IMPORTED)
    set_target_properties(OpenSSL::SSL PROPERTIES IMPORTED_LOCATION "${prefix}/lib/libssl.a" INTERFACE_INCLUDE_DIRECTORIES "${prefix}/include" INTERFACE_LINK_LIBRARIES "OpenSSL::Crypto")
    if(CF_BOOST_SOURCE)
        include(cmake/LocalBoost.cmake)
        cf_local_boost("${CF_BOOST_SOURCE}")
        set(json_target cf_json_local)
    else()
        message(FATAL_ERROR "CF_BOOST_SOURCE is required; run tools/prepare-native first")
    endif()
else()
    find_package(JNI REQUIRED)
    find_package(OpenSSL REQUIRED)
    if(CF_BOOST_SOURCE)
        include(cmake/LocalBoost.cmake)
        cf_local_boost("${CF_BOOST_SOURCE}")
        set(json_target cf_json_local)
    else()
        find_package(Boost 1.83 REQUIRED)
        add_library(cf_json_host STATIC src/boost_json_host.cpp)
        target_include_directories(cf_json_host SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
        set(json_target cf_json_host)
    endif()
endif()

add_library(cf_tunnel SHARED src/tunnel_engine.cpp src/jni_bridge.cpp)
target_include_directories(cf_tunnel PRIVATE src ${JNI_INCLUDE_DIRS})
if(ANDROID)
    if(CF_BOOST_SOURCE)
        target_link_libraries(cf_tunnel PRIVATE cf_boost_headers)
    else()
        target_link_libraries(cf_tunnel PRIVATE Boost::asio Boost::beast)
    endif()
else()
    target_include_directories(cf_tunnel SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
endif()
target_link_libraries(cf_tunnel PRIVATE ${json_target} OpenSSL::SSL OpenSSL::Crypto)
target_compile_options(cf_tunnel PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
    target_link_libraries(cf_tunnel PRIVATE log android dl)
    target_link_options(cf_tunnel PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--exclude-libs,ALL)
endif()
